-- ===================================================================== -- SHK PRIVATE - SQL MỚI -- Chạy toàn bộ file này 1 lần trong Supabase > SQL Editor. -- Cách hoạt động: mỗi lần nhập là 1 dòng trong rc_records. -- rc_best = kỷ lục nhanh nhất của mỗi tay đua trên mỗi map -- rc_current = top 5 thời gian thấp nhất mỗi map (tự xếp hạng + tự tính điểm) -- Dữ liệu top 5 hiện tại của bản cũ được giữ lại và chuyển sang bản mới. -- Tài khoản (rc_profiles) được giữ nguyên. -- ===================================================================== create extension if not exists unaccent with schema extensions; -- ---------- 1. Sao lưu top 5 cũ (nếu có) ---------- do $$ begin if to_regclass('public.rc_current') is not null and to_regclass('public.rc_maps') is not null and to_regclass('public.rc_racers') is not null and to_regclass('public.bk_cur') is null then execute 'create table public.bk_maps as select name, slug, difficulty, image_url from public.rc_maps'; execute 'create table public.bk_racers as select name, slug from public.rc_racers'; execute 'create table public.bk_cur as select m.slug as map_slug, r.slug as racer_slug, c.time_ms, c.car, c.pet, c.posted_on from public.rc_current c join public.rc_maps m on m.id = c.map_id join public.rc_racers r on r.id = c.racer_id'; end if; end $$; -- ---------- 2. Xóa cấu trúc cũ ---------- do $$ declare f record; k "char"; begin for f in select oid::regprocedure as p from pg_proc where pronamespace = 'public'::regnamespace and proname in ('rc_save_snapshot','rc_delete_entry','rc_add_record') loop execute 'drop function ' || f.p || ' cascade'; end loop; for k in select relkind from pg_class where relnamespace = 'public'::regnamespace and relname = 'rc_current' loop if k = 'v' then execute 'drop view public.rc_current cascade'; else execute 'drop table public.rc_current cascade'; end if; end loop; end $$; drop view if exists public.rc_best cascade; drop table if exists public.rc_entries cascade; drop table if exists public.rc_snapshots cascade; drop table if exists public.rc_records cascade; drop table if exists public.rc_racers cascade; drop table if exists public.rc_maps cascade; -- ---------- 3. Bảng ---------- create table if not exists public.rc_profiles( id uuid primary key references auth.users(id) on delete cascade, username text unique not null, role text not null default 'viewer' check (role in ('viewer','admin')), created_at timestamptz not null default now() ); create table public.rc_maps( id uuid primary key default gen_random_uuid(), name text not null, slug text not null unique, difficulty int check (difficulty between 1 and 7), image_url text, created_at timestamptz not null default now() ); create table public.rc_racers( id uuid primary key default gen_random_uuid(), name text not null, slug text not null unique, created_at timestamptz not null default now() ); create table public.rc_records( id uuid primary key default gen_random_uuid(), map_id uuid not null references public.rc_maps(id) on delete cascade, racer_id uuid not null references public.rc_racers(id) on delete cascade, time_ms int not null check (time_ms > 0), car text not null default '', pet text not null default '', posted_on date, created_at timestamptz not null default now() ); create index rc_records_map_idx on public.rc_records(map_id, time_ms); create index rc_records_racer_idx on public.rc_records(racer_id); -- ---------- 4. View: kỷ lục tốt nhất + top 5 ---------- create view public.rc_best with (security_invoker = on) as select distinct on (map_id, racer_id) id, map_id, racer_id, time_ms, car, pet, posted_on, created_at from public.rc_records order by map_id, racer_id, time_ms, created_at; create view public.rc_current with (security_invoker = on) as select id, map_id, racer_id, time_ms, car, pet, posted_on, created_at, rk::int as rank, (6 - rk)::int as points from ( select b.*, row_number() over (partition by b.map_id order by b.time_ms, b.created_at) as rk from public.rc_best b ) x where rk <= 5; -- ---------- 5. Phân quyền: is_admin() ---------- create or replace function public.is_admin() returns boolean language sql stable security definer set search_path = public as $$ select exists (select 1 from public.rc_profiles where id = auth.uid() and role = 'admin') $$; revoke all on function public.is_admin() from public, anon; grant execute on function public.is_admin() to authenticated; -- ---------- 6. Tự tạo profile khi đăng ký ---------- create or replace function public.rc_new_user() returns trigger language plpgsql security definer set search_path = public as $$ begin insert into public.rc_profiles(id, username, role) values (new.id, split_part(new.email, '@', 1), 'viewer') on conflict (id) do nothing; return new; end $$; drop trigger if exists rc_on_signup on auth.users; create trigger rc_on_signup after insert on auth.users for each row execute function public.rc_new_user(); -- tạo profile cho tài khoản nào còn thiếu insert into public.rc_profiles(id, username, role) select u.id, split_part(u.email, '@', 1), 'viewer' from auth.users u where not exists (select 1 from public.rc_profiles p where p.id = u.id) on conflict do nothing; -- ---------- 7. Hàm nhập kỷ lục (chỉ admin) ---------- create or replace function public.rc_add_record( p_map uuid, p_name text, p_ms int, p_car text, p_pet text, p_date date ) returns jsonb language plpgsql security definer set search_path = public, extensions as $$ declare v_name text := regexp_replace(trim(coalesce(p_name,'')), '\s+', ' ', 'g'); v_slug text; v_racer uuid; v_new boolean := false; v_best int; v_rank int; begin if not public.is_admin() then raise exception 'Bạn không có quyền admin.'; end if; if v_name = '' then raise exception 'Cần nhập tên tay đua.'; end if; if p_ms is null or p_ms <= 0 then raise exception 'Kỷ lục không hợp lệ.'; end if; if not exists (select 1 from public.rc_maps where id = p_map) then raise exception 'Map không tồn tại.'; end if; v_slug := trim(both '-' from regexp_replace(lower(unaccent(v_name)), '[^a-z0-9]+', '-', 'g')); if v_slug = '' then v_slug := 'x' || substr(md5(v_name), 1, 8); end if; select id into v_racer from public.rc_racers where slug = v_slug; if v_racer is null then insert into public.rc_racers(name, slug) values (v_name, v_slug) returning id into v_racer; v_new := true; end if; select min(time_ms) into v_best from public.rc_records where map_id = p_map and racer_id = v_racer; if v_best is not null and p_ms >= v_best then return jsonb_build_object('status','slower','best',v_best, 'racer',(select name from public.rc_racers where id = v_racer)); end if; insert into public.rc_records(map_id, racer_id, time_ms, car, pet, posted_on) values (p_map, v_racer, p_ms, coalesce(trim(p_car),''), coalesce(trim(p_pet),''), p_date); -- hạng = 1 + số tay đua khác có kỷ lục tốt nhất nhanh hơn hoặc bằng select count(*) + 1 into v_rank from (select racer_id, min(time_ms) as t from public.rc_records where map_id = p_map and racer_id <> v_racer group by racer_id) x where x.t <= p_ms; return jsonb_build_object( 'status', case when v_rank <= 5 then 'top' else 'out' end, 'rank', v_rank, 'new_racer', v_new, 'racer', (select name from public.rc_racers where id = v_racer)); end $$; revoke all on function public.rc_add_record(uuid,text,int,text,text,date) from public, anon; grant execute on function public.rc_add_record(uuid,text,int,text,text,date) to authenticated; -- ---------- 8. Khôi phục dữ liệu cũ ---------- do $$ begin if to_regclass('public.bk_maps') is not null then insert into public.rc_maps(name, slug, difficulty, image_url) select name, slug, difficulty, image_url from public.bk_maps on conflict (slug) do nothing; insert into public.rc_racers(name, slug) select name, slug from public.bk_racers on conflict (slug) do nothing; insert into public.rc_records(map_id, racer_id, time_ms, car, pet, posted_on) select m.id, r.id, b.time_ms, coalesce(b.car,''), coalesce(b.pet,''), b.posted_on from public.bk_cur b join public.rc_maps m on m.slug = b.map_slug join public.rc_racers r on r.slug = b.racer_slug; end if; end $$; -- ---------- 9. RLS ---------- alter table public.rc_profiles enable row level security; alter table public.rc_maps enable row level security; alter table public.rc_racers enable row level security; alter table public.rc_records enable row level security; do $$ declare p record; begin for p in select schemaname, tablename, policyname from pg_policies where schemaname = 'public' and tablename in ('rc_profiles','rc_maps','rc_racers','rc_records') loop execute format('drop policy %I on %I.%I', p.policyname, p.schemaname, p.tablename); end loop; end $$; -- profiles: xem của mình (admin xem tất cả), chỉ admin đổi quyền create policy prof_select on public.rc_profiles for select to authenticated using (id = auth.uid() or public.is_admin()); create policy prof_update on public.rc_profiles for update to authenticated using (public.is_admin()) with check (public.is_admin()); -- maps / racers / records: ai đăng nhập cũng xem được, chỉ admin sửa create policy maps_select on public.rc_maps for select to authenticated using (true); create policy maps_write on public.rc_maps for all to authenticated using (public.is_admin()) with check (public.is_admin()); create policy racers_select on public.rc_racers for select to authenticated using (true); create policy racers_write on public.rc_racers for all to authenticated using (public.is_admin()) with check (public.is_admin()); create policy rec_select on public.rc_records for select to authenticated using (true); create policy rec_write on public.rc_records for all to authenticated using (public.is_admin()) with check (public.is_admin()); grant select on public.rc_best, public.rc_current to authenticated; -- ---------- 10.
